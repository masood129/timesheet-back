CREATE TABLE Projects (
  Id INT PRIMARY KEY IDENTITY(1,1),
  ProjectName NVARCHAR(100),
  securityLevel INT ,
)

CREATE TABLE DailyDetails (
    Date DATE NOT NULL,
    UserId INT NOT NULL,
    ArrivalTime TIME,
    LeaveTime TIME,
    LeaveType NVARCHAR(50),
    PersonalTime INT,
    Description NVARCHAR(MAX),
    GoCost INT,
    ReturnCost INT,
    PersonalCarCost INT,
    PRIMARY KEY (Date, UserId),
    FOREIGN KEY (UserId) REFERENCES Users(Id)
);


CREATE TABLE DailyProjectTasks (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Date DATE NOT NULL,
    UserId INT NOT NULL,
    ProjectId INT NOT NULL,
    Duration INT,
    Description NVARCHAR(MAX),
    FOREIGN KEY (Date, UserId) REFERENCES DailyDetails(Date, UserId),
    FOREIGN KEY (ProjectId) REFERENCES Projects(Id)
);



-- اطمینان از وجود جدول DailyDetails
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'DailyDetails' AND SCHEMA_NAME(schema_id) = 'dbo')
BEGIN
    RAISERROR ('Table DailyDetails does not exist in schema [flutter].[dbo]. Please create it first.', 16, 1);
    RETURN;
END

-- اطمینان از وجود جدول Projects
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'Projects' AND SCHEMA_NAME(schema_id) = 'dbo')
BEGIN
    RAISERROR ('Table Projects does not exist in schema [flutter].[dbo]. Please create it first.', 16, 1);
    RETURN;
END

-- ایجاد جدول جدید برای هزینه‌های خودرو شخصی
CREATE TABLE [flutter].[dbo].[DailyPersonalCarCosts] (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Date DATE NOT NULL,
    UserId INT NOT NULL,
    ProjectId INT NOT NULL,
    Cost INT NOT NULL,
    Description NVARCHAR(MAX),
    CONSTRAINT FK_DailyPersonalCarCosts_DailyDetails FOREIGN KEY (Date, UserId) 
        REFERENCES [flutter].[dbo].[DailyDetails](Date, UserId),
    CONSTRAINT FK_DailyPersonalCarCosts_Projects FOREIGN KEY (ProjectId) 
        REFERENCES [flutter].[dbo].[Projects](Id)
);
