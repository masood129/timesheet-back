-- ایجاد جدول Projects (بدون تغییر از فایل شما) [cite: 1]
CREATE TABLE Projects (
  Id INT PRIMARY KEY IDENTITY(1,1),
  ProjectName NVARCHAR(100),
  securityLevel INT
);

-- ایجاد جدول Users (جدید - اگر قبلاً نداشته‌اید)
-- این جدول برای نگهداری اطلاعات کاربران و نقش آنها لازم است
CREATE TABLE Users (
    UserId INT PRIMARY KEY IDENTITY(1,1),
    Username NVARCHAR(100) NOT NULL UNIQUE,
    PasswordHash NVARCHAR(MAX) NOT NULL, -- همیشه پسوردها را هش کنید
    Role NVARCHAR(50) NOT NULL DEFAULT 'user' -- 'user', 'admin'
    -- سایر فیلدهای کاربر مانند نام، ایمیل و ...
);

-- ایجاد جدول DailyDetails (بدون تغییر زیاد از فایل شما) [cite: 1]
CREATE TABLE DailyDetails (
    Date DATE,
    UserId INT,
    ArrivalTime NVARCHAR(8), -- HH:MM:SS
    LeaveTime NVARCHAR(8),   -- HH:MM:SS
    LeaveType NVARCHAR(50),
    PersonalTime INT,        -- In minutes
    Description NVARCHAR(MAX),
    GoCost INT,
    ReturnCost INT,
    PRIMARY KEY (Date, UserId),
    FOREIGN KEY (UserId) REFERENCES Users(UserId) -- ارتباط با جدول کاربران
);

-- ایجاد جدول DailyProjectTasks (بدون تغییر از فایل شما) [cite: 2]
CREATE TABLE DailyProjectTasks (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Date DATE NOT NULL,
    UserId INT NOT NULL,
    ProjectId INT NOT NULL,
    Duration INT,            -- In minutes
    Description NVARCHAR(MAX),
    FOREIGN KEY (Date, UserId) REFERENCES DailyDetails(Date, UserId),
    FOREIGN KEY (ProjectId) REFERENCES Projects(Id)
);

-- ایجاد جدول DailyPersonalCarCosts (با افزودن Kilometers از فایل شما)
CREATE TABLE DailyPersonalCarCosts ( -- نام schema از فایل شما حذف شد برای سادگی، در صورت نیاز اضافه کنید
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Date DATE NOT NULL,
    UserId INT NOT NULL,
    ProjectId INT NOT NULL,
    Cost INT NOT NULL,
    Kilometers INT NULL, -- بر اساس ALTER TABLE شما [cite: 6]
    Description NVARCHAR(MAX),
    FOREIGN KEY (Date, UserId) REFERENCES DailyDetails(Date, UserId),
    FOREIGN KEY (ProjectId) REFERENCES Projects(Id)
);

-- جدول Holidays (جدید)
CREATE TABLE Holidays (
    HolidayId INT PRIMARY KEY IDENTITY(1,1),
    Date DATE NOT NULL UNIQUE,
    Name NVARCHAR(255) NOT NULL,
    HolidayType NVARCHAR(100), -- e.g., 'رسمی', 'شرکتی', 'مناسبت'
    IsWorkDay BIT NOT NULL DEFAULT 0 -- 0 for holiday, 1 for a working event day
);

-- جدول UserContractHours (جدید)
CREATE TABLE UserContractHours (
    ContractId INT PRIMARY KEY IDENTITY(1,1),
    UserId INT NOT NULL UNIQUE, -- هر کاربر یک مجموعه ساعات قرارداد فعال دارد
    ContractArrivalTime NVARCHAR(8) NOT NULL, -- HH:MM:SS
    ContractLeaveTime NVARCHAR(8) NOT NULL,  -- HH:MM:SS
    -- EffectiveDate DATE NOT NULL, -- برای تاریخچه تغییرات، فعلا ساده نگه میداریم
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);
